38  /**
39   * <div>
40   * Checks for fully qualified type references that are unnecessary because the
41   * type could simply be imported and referenced by its simple name.
42   * </div>
43   *
44   * <p>
45   * A fully qualified type reference, such as {@code java.util.Map}, is reported
46   * as unnecessary unless another type with the same simple name is also used in
47   * the file. For example, {@code java.util.Map} is allowed only when a different
48   * {@code Map} (e.g. {@code com.example.Map}, a locally declared {@code Map}, or
49   * a same-package or {@code java.lang} {@code Map}) is also referenced, since in
50   * that case qualification is required to tell the two types apart.
51   * </p>
52   *
53   * <p>
54   * Using fully qualified names where they are not needed reduces readability and
55   * is inconsistent with common Java style conventions, which favor importing the type instead.
56   * </p>
57   *
58   * <p>
59   * The following limitations apply:
60   * </p>
61   * <ul>
62   * <li>
63   * To distinguish a package qualified type (such as {@code java.util.Map}) from a
64   * reference to a nested type (such as {@code Map.Entry}), the check relies on the
65   * common convention that package names start with a lowercase letter. A qualified
66   * reference is treated as fully qualified only when the identifier preceding the
67   * simple name starts with a lowercase letter.
68   * </li>
69   * <li>
70   * References to nested types via an enclosing type (for example
71   * {@code java.util.Map.Entry}) and fully qualified annotations
72   * (for example {@code @java.lang.Override}) are not reported.
73   * </li>
74   * <li>
75   * On-demand (wildcard) imports, such as {@code import java.util.*;}, are checked
76   * against classes reachable on this check's own class loader. This reliably
77   * detects collisions from JDK packages (for example a fully qualified reference
78   * to a third-party {@code Observer} type being left alone because
79   * {@code import java.util.*;} also brings a different {@code java.util.Observer}
80   * into scope), but on-demand imports of third-party or project-local packages
81   * cannot be resolved this way and may still be flagged incorrectly.
82   * </li>
83   * </ul>
84   *
85   * @since 14.1.0
86   */
87  @FileStatefulCheck
88  public class UnnecessaryFullyQualifiedTypeCheck extends AbstractCheck {
89  
90      /**
91       * A key is pointing to the warning message text in "messages.properties" file.
92       */
93      public static final String MSG_KEY = "unnecessary.fully.qualified.type";
94  
95      /** Array declarator markers that may trail a type name. */
96      private static final String ARRAY_MARKERS = "[]";
97  
98      /** Marker used to represent a locally declared type in the identity set. */
99      private static final String DECLARED_MARKER = "<declared> ";
100 
101     /** Marker used to represent an unqualified type reference in the identity set. */
102     private static final String UNQUALIFIED_MARKER = "<unqualified> ";
103 
104     /** The {@code java.lang} package, types of which never need to be qualified. */
105     private static final String JAVA_LANG_PACKAGE = "java.lang";
106 
107     /** Suffix identifying an on-demand (wildcard) import, such as {@code java.util.*}. */
108     private static final String ON_DEMAND_IMPORT_SUFFIX = ".*";
109 
110     /** Separator used between components of a fully qualified name. */
111     private static final String DOT = ".";
112 
113     /**
114      * Tokens whose direct child is a type reference. A {@link TokenTypes#DOT} or
115      * {@link TokenTypes#IDENT} is considered to be in type context when its parent
116      * is one of these tokens.
117      */
118     private static final int[] TYPE_CONTEXT_PARENTS = {
119         TokenTypes.TYPE,
120         TokenTypes.LITERAL_NEW,
121         TokenTypes.EXTENDS_CLAUSE,
122         TokenTypes.IMPLEMENTS_CLAUSE,
123         TokenTypes.LITERAL_THROWS,
124         TokenTypes.TYPE_ARGUMENT,
125     };
126 
127     /**
128      * Tokens that declare a type. An {@link TokenTypes#IDENT} whose parent is one of
129      * these is the name of a type declared in the file.
130      */
131     private static final int[] TYPE_DECLARATION_PARENTS = {
132         TokenTypes.CLASS_DEF,
133         TokenTypes.INTERFACE_DEF,
134         TokenTypes.ENUM_DEF,
135         TokenTypes.ANNOTATION_DEF,
136         TokenTypes.RECORD_DEF,
137     };
138 
139     /** Maps the simple name of each single-type import to its fully qualified name. */
140     private final Map<String, String> importedTypes = new HashMap<>();
141 
142     /** Simple names of the types declared in the file. */
143     private final Set<String> declaredTypes = new HashSet<>();
144 
145     /** Simple names of the types referenced without qualification in the file. */
146     private final Set<String> simpleReferences = new HashSet<>();
147 
148     /** Fully qualified type references collected while traversing the file. */
149     private final List<QualifiedTypeReference> qualifiedReferences = new ArrayList<>();
150 
151     /**
152      * Packages imported on-demand (wildcard), e.g. {@code java.util} for
153      * {@code import java.util.*;}.
154      */
155     private final Set<String> onDemandImportPackages = new HashSet<>();
156 
157     /** The package the file belongs to, or {@code null} for the default package. */
158     @Nullable
159     private String packageName;
160 
161     /**
162      * Creates new {@code UnnecessaryFullyQualifiedTypeCheck} instance.
163      */
164     public UnnecessaryFullyQualifiedTypeCheck() {
165         // no code by default
166     }
167 
168     @Override
169     public int[] getDefaultTokens() {
170         return getRequiredTokens();
171     }
172 
173     @Override
174     public int[] getAcceptableTokens() {
175         return getRequiredTokens();
176     }
177 
178     @Override
179     public int[] getRequiredTokens() {
180         return new int[] {
181             TokenTypes.PACKAGE_DEF,
182             TokenTypes.IMPORT,
183             TokenTypes.DOT,
184             TokenTypes.IDENT,
185         };
186     }
187 
188     @Override
189     public void beginTree(DetailAST rootAST) {
190         packageName = null;
191         onDemandImportPackages.clear();
192         importedTypes.clear();
193         declaredTypes.clear();
194         simpleReferences.clear();
195         qualifiedReferences.clear();
196     }
197 
198     @Override
199     public void visitToken(DetailAST ast) {
200         switch (ast.getType()) {
201             case TokenTypes.PACKAGE_DEF -> processPackage(ast);
202             case TokenTypes.IMPORT -> processImport(ast);
203             case TokenTypes.DOT -> processDot(ast);
204             default -> processIdent(ast);
205         }
206     }
207 
208     @Override
209     public void finishTree(DetailAST rootAST) {
210         for (QualifiedTypeReference reference : qualifiedReferences) {
211             if (isUnnecessary(reference.simpleName())) {
212                 log(reference.detailAst(), MSG_KEY, reference.fullName());
213             }
214         }
215     }
216 
217     /**
218      * Records the package the file belongs to.
219      *
220      * @param ast the {@link TokenTypes#PACKAGE_DEF} node
221      */
222     private void processPackage(DetailAST ast) {
223         packageName = FullIdent.createFullIdent(
224                 ast.getLastChild().getPreviousSibling()).getText();
225     }
226 
227     /**
228      * Records an import by its simple name so that later references can be resolved.
229      * On-demand (wildcard) imports do not have a single simple name, so the
230      * imported package is recorded separately instead.
231      *
232      * @param ast the {@link TokenTypes#IMPORT} node
233      */
234     private void processImport(DetailAST ast) {
235         final String importText = FullIdent.createFullIdentBelow(ast).getText();
236         if (importText.endsWith(ON_DEMAND_IMPORT_SUFFIX)) {
237             onDemandImportPackages.add(
238                 importText.substring(0, importText.length() - ON_DEMAND_IMPORT_SUFFIX.length()));
239         }
240         else {
241             importedTypes.put(extractSimpleName(importText), importText);
242         }
243     }
244 
245     /**
246      * Collects a fully qualified type reference if the given {@link TokenTypes#DOT}
247      * node represents one.
248      *
249      * @param ast the {@code TokenTypes#DOT} node
250      */
251     private void processDot(DetailAST ast) {
252         if (isInTypeContext(ast)) {
253             final String fullName = buildFullName(ast).replace(ARRAY_MARKERS, "");
254             if (isFullyQualified(fullName)) {
255                 qualifiedReferences.add(
256                         new QualifiedTypeReference(ast, fullName, extractSimpleName(fullName)));
257             }
258         }
259     }
260 
261     /**
262      * Reconstructs the fully qualified name for a {@code TokenTypes#DOT} node,
263      * handling the extra child produced by an embedded type-use annotation.
264      *
265      * @param ast the {@code TokenTypes#DOT} node
266      * @return the fully qualified name the node represents
267      */
268     private static String buildFullName(DetailAST ast) {
269         final DetailAST qualifier = ast.getFirstChild();
270         final DetailAST secondChild = qualifier.getNextSibling();
271         final String result;
272         if (secondChild.getType() == TokenTypes.ANNOTATIONS) {
273             final DetailAST simpleNameIdent = secondChild.getNextSibling();
274             result = FullIdent.createFullIdent(qualifier).getText()
275                     + DOT + simpleNameIdent.getText();
276         }
277         else {
278             result = FullIdent.createFullIdent(ast).getText();
279         }
280         return result;
281     }
282 
283     /**
284      * Records a reference made through a simple name. The name is either an
285      * unqualified type reference or the name of a type declared in the file.
286      *
287      * @param ast the {@link TokenTypes#IDENT} node
288      */
289     private void processIdent(DetailAST ast) {
290         final DetailAST parent = ast.getParent();
291         if (TokenUtil.isOfType(parent, TYPE_CONTEXT_PARENTS)) {
292             simpleReferences.add(ast.getText());
293         }
294         else if (TokenUtil.isOfType(parent, TYPE_DECLARATION_PARENTS)) {
295             declaredTypes.add(ast.getText());
296         }
297     }
298 
299     /**
300      * Determines whether a type reference with the given simple name is unnecessary.
301      * It is unnecessary when this fully qualified reference is the only type with
302      * that simple name used in the file, so it could be imported and referenced by
303      * its simple name without introducing ambiguity.
304      *
305      * @param simpleName the simple name of the type
306      * @return {@code true} if the fully qualified reference is unnecessary
307      */
308     private boolean isUnnecessary(String simpleName) {
309         return collectIdentities(simpleName).size() == 1;
310     }
311 
312     /**
313      * Collects every distinct identity claiming the given simple name in the file:
314      * a direct import, each fully qualified reference sharing that simple name, a
315      * locally declared type, an unqualified reference resolving elsewhere, and a
316      * type reachable via an on-demand import. A simple name is unambiguous, and
317      * therefore its fully qualified reference unnecessary, exactly when this set
318      * has a single member.
319      *
320      * @param simpleName the simple name to collect identities for
321      * @return the set of distinct identities claiming this simple name
322      */
323     private Set<String> collectIdentities(String simpleName) {
324         final Set<String> identities = new HashSet<>();
325         final String imported = importedTypes.get(simpleName);
326         if (imported != null) {
327             identities.add(imported);
328         }
329         if (declaredTypes.contains(simpleName)) {
330             identities.add(DECLARED_MARKER + simpleName);
331         }
332         addQualifiedReferenceIdentities(simpleName, identities);
333         if (imported == null) {
334             final boolean unqualifiedResolvedToReference =
335                     resolvesToQualifiedReference(simpleName);
336             addImplicitIdentities(simpleName, unqualifiedResolvedToReference, identities);
337         }
338         return identities;
339     }
340 
341     /**
342      * Adds an identity for each fully qualified reference sharing the given simple name.
343      *
344      * @param simpleName the simple name to match references against
345      * @param identities the identity set to add to
346      */
347     private void addQualifiedReferenceIdentities(String simpleName, Set<String> identities) {
348         for (QualifiedTypeReference reference : qualifiedReferences) {
349             if (reference.simpleName().equals(simpleName)) {
350                 identities.add(reference.fullName());
351             }
352         }
353     }
354 
355     /**
356      * Checks whether any fully qualified reference with the given simple name
357      * resolves to the same type as an unqualified reference would.
358      *
359      * @param simpleName the simple name to check
360      * @return {@code true} if an unqualified reference resolves to a matching qualified type
361      */
362     private boolean resolvesToQualifiedReference(String simpleName) {
363         boolean resolved = false;
364         for (QualifiedTypeReference reference : qualifiedReferences) {
365             if (reference.simpleName().equals(simpleName)
366                     && resolvesUnqualified(reference.fullName())) {
367                 resolved = true;
368                 break;
369             }
370         }
371         return resolved;
372     }
373 
374     /**
375      * Adds identities that are not established by an explicit import: an
376      * unqualified reference resolving to a same-package or {@code java.lang} type,
377      * and a type reachable via an on-demand import. Only relevant when no direct
378      * import of this simple name exists, since a direct import always takes
379      * priority over an on-demand import.
380      *
381      * @param simpleName the simple name to check
382      * @param unqualifiedResolvedToReference whether a same-simple-name qualified
383      *     reference already resolves to what an unqualified use would mean
384      * @param identities the identity set to add to
385      */
386     private void addImplicitIdentities(String simpleName,
387                                        boolean unqualifiedResolvedToReference,
388                                        Set<String> identities) {
389         if (!unqualifiedResolvedToReference && simpleReferences.contains(simpleName)) {
390             identities.add(UNQUALIFIED_MARKER + simpleName);
391         }
392         final String onDemandFullName = resolveViaOnDemandImport(simpleName);
393         if (onDemandFullName != null) {
394             identities.add(onDemandFullName);
395         }
396     }
397 
398     /**
399      * Determines whether a reference by simple name (without import or local
400      * declaration) would resolve to the type denoted by the given fully qualified
401      * name. This is the case when that name belongs to the {@code java.lang}
402      * package or to the package the file itself belongs to.
403      *
404      * @param fullName a fully qualified type name
405      * @return {@code true} if an unqualified reference would resolve to it
406      */
407     private boolean resolvesUnqualified(String fullName) {
408         final String qualifier = fullName.substring(0, fullName.lastIndexOf('.'));
409         return JAVA_LANG_PACKAGE.equals(qualifier) || qualifier.equals(packageName);
410     }
411 
412     /**
413      * Resolves a simple name against the file's on-demand imports, returning the
414      * fully qualified name of the class it resolves to, if any. Resolution is
415      * attempted only against classes reachable on this check's own class loader,
416      * so it is reliable for JDK packages but cannot detect collisions caused by
417      * third-party or project-local on-demand imports.
418      *
419      * @param simpleName the simple name to look for
420      * @return the fully qualified name the simple name resolves to via an
421      *     on-demand import, or {@code null} if none of the on-demand imported
422      *     packages contain a class of that name
423      */
424     @Nullable
425     private String resolveViaOnDemandImport(String simpleName) {
426         String resolved = null;
427         for (String onDemandPackage : onDemandImportPackages) {
428             try {
429                 final String candidate = onDemandPackage + DOT + simpleName;
430                 Class.forName(candidate, false, null);
431                 resolved = candidate;
432                 break;
433             }
434             catch (ClassNotFoundException | LinkageError ignored) {
435                 // this package does not contain a class of this name; keep checking
436             }
437         }
438         return resolved;
439     }

441     /**
442      * Checks whether the given node is a direct child of a type context token.
443      *
444      * @param ast the node to check
445      * @return {@code true} if the node is in type context
446      */
447     private static boolean isInTypeContext(DetailAST ast) {
448         return TokenUtil.isOfType(ast.getParent(), TYPE_CONTEXT_PARENTS);
449     }
450 
451     /**
452      * Determines whether the given qualified name is a package qualified (fully
453      * qualified) type reference, as opposed to a reference to a nested type. The
454      * check relies on the convention that package names start with a lowercase
455      * letter, so the name is treated as fully qualified when the identifier
456      * preceding the simple name starts with a lowercase letter.
457      *
458      * @param name a qualified name containing at least one dot, as produced from
459      *     a {@link TokenTypes#DOT} node
460      * @return {@code true} if the name is a fully qualified type reference
461      */
462     private static boolean isFullyQualified(String name) {
463         final int simpleNameStart = name.lastIndexOf('.');
464         final int qualifierStart = name.lastIndexOf('.', simpleNameStart - 1);
465         final char firstQualifierChar = name.charAt(qualifierStart + 1);
466         return Character.isLowerCase(firstQualifierChar);
467     }
468 
469     /**
470      * Extracts the simple name from a qualified name.
471      *
472      * @param name a qualified name
473      * @return the part of the name after the last dot
474      */
475     private static String extractSimpleName(String name) {
476         return name.substring(name.lastIndexOf('.') + 1);
477     }

487     }
