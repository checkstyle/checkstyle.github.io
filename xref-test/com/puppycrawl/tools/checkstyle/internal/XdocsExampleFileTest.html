63      @Test
64      public void testAllCheckPropertiesAreUsedInXdocsExamples() throws Exception {
65          final Map<String, Set<String>> usedPropertiesByCheck =
66              XdocUtil.extractUsedPropertiesFromXdocsExamples();
67          final List<String> failures = new ArrayList<>();
68  
69          for (Class<?> checkClass : CheckUtil.getCheckstyleChecks()) {
70              final String checkSimpleName = checkClass.getSimpleName();
71  
72              final Set<String> definedProperties = Arrays.stream(
73                      PropertyUtils.getPropertyDescriptors(checkClass))
74                  .filter(descriptor -> descriptor.getWriteMethod() != null)
75                  .map(PropertyDescriptor::getName)
76                  .filter(property -> !COMMON_PROPERTIES.contains(property))
77                  .collect(Collectors.toUnmodifiableSet());
78  
79              final Set<String> usedProperties =
80                  usedPropertiesByCheck.getOrDefault(checkSimpleName, Collections.emptySet());
81  
82              for (String property : definedProperties) {
83                  if (!usedProperties.contains(property)) {
84                      failures.add("Missing property in xdoc: '"
85                              + property + "' of " + checkSimpleName);
86                  }
87              }
88          }
89          if (!failures.isEmpty()) {
90              assertWithMessage("Xdocs are missing properties:\n" + String.join("\n", failures))
91                      .fail();
92          }
93      }

95      @Test
96      public void testAllExampleFilesHaveCorrespondingTestMethods() throws Exception {
97          final Path examplesResources = Path.of("src/xdocs-examples/resources");
98          final Path examplesNonCompilable = Path.of("src/xdocs-examples/resources-noncompilable");
99          final Path examplesTestRoot = Path.of(
100             "src/xdocs-examples/java/com/puppycrawl/tools/checkstyle/checks");
101         final List<String> failures = new ArrayList<>();
102 
103         try (Stream<Path> testFiles = Files.walk(examplesTestRoot)) {
104             testFiles
105                 .filter(path -> path.toString().endsWith("ExamplesTest.java"))
106                 .forEach(testFile -> {
107                     try {
108                         scanFile(testFile, examplesResources, examplesNonCompilable, failures);
109                     }
110                     catch (IOException exception) {
111                         throw new IllegalStateException("Error processing: "
112                                      + testFile, exception);
113                     }
114                 });
115         }
116         if (!failures.isEmpty()) {
117             assertWithMessage("Example files are missing corresponding test methods:\n"
118                     + String.join("\n", failures))
119                     .fail();
120         }
121     }

123     @Test
124     public void testAllExampleFilesAreReferencedInXdocs() throws Exception {
125         final Set<String> referencedPaths = collectReferencedExamplePaths();
126         final Path xdocsExamplesBase = Path.of("src/xdocs-examples");
127         final List<Path> exampleRoots = List.of(
128             xdocsExamplesBase.resolve("resources"),
129             xdocsExamplesBase.resolve("resources-noncompilable")
130         );
131         final List<String> failures = new ArrayList<>();
132 
133         for (Path root : exampleRoots) {
134             if (Files.exists(root)) {
135                 try (Stream<Path> paths = Files.walk(root)) {
136                     paths
137                         .filter(path -> {
138                             final String fileName = path.getFileName().toString();
139                             return Files.isRegularFile(path)
140                                 && (fileName.startsWith("Example")
141                                     || fileName.startsWith("UseCase"));
142                         })
143                         .forEach(exampleFile -> {
144                             final String relative = xdocsExamplesBase
145                                 .relativize(exampleFile)
146                                 .toString()
147                                 .replace(File.separatorChar, '/');
148                             if (!referencedPaths.contains(relative)) {
149                                 failures.add(relative);
150                             }
151                         });
152                 }
153             }
154         }
155 
156         if (!failures.isEmpty()) {
157             assertWithMessage(
158                 "The following example files are not referenced in any xml.template file:\n"
159                     + String.join("\n", failures))
160                 .fail();
161         }
162     }

164     private static Set<String> collectReferencedExamplePaths() throws Exception {
165         final Set<String> referenced = new HashSet<>();
166 
167         for (Path template : XdocUtil.getXdocsTemplatesFilePaths()) {
168             final String input = Files.readString(template);
169             final Document document = XmlUtil.getRawXml(template.toString(), input, input);
170             final NodeList macros = document.getElementsByTagName("macro");
171 
172             for (int idx = 0; idx < macros.getLength(); idx++) {
173                 final Element macro = (Element) macros.item(idx);
174                 if ("example".equals(macro.getAttribute("name"))) {
175                     final String path = getMacroParamValue(macro, "path");
176                     if (path != null && !path.isEmpty()) {
177                         referenced.add(normalizePath(path));
178                     }
179                 }
180             }
181         }
182         return referenced;
183     }
184 
185     private static String getMacroParamValue(Element macro, String paramName) {
186         String result = null;
187         final NodeList params = macro.getElementsByTagName("param");
188 
189         for (int idx = 0; idx < params.getLength(); idx++) {
190             final Element param = (Element) params.item(idx);
191             if (paramName.equals(param.getAttribute("name"))) {
192                 result = param.getAttribute("value");
193                 break;
194             }
195         }
196         return result;
197     }
198 
199     private static String normalizePath(String path) {
200         String result = path;
201         if (result.startsWith("/")) {
202             result = result.substring(1);
203         }
204         return result;
205     }

207     private static void scanFile(Path testFile, Path examplesResources, Path examplesNonCompilable,
208             List<String> failures)
209             throws IOException {
210         final String testContent = Files.readString(testFile);
211 
212         final String className = Path.of("src/xdocs-examples/java").toAbsolutePath()
213                 .relativize(testFile.toAbsolutePath()).toString()
214                 .replace(File.separator, ".")
215                 .replaceFirst("\\.java$", "");
216 
217         try {
218             final Class<?> testClass = Class.forName(className);
219             final AbstractPathTestSupport instance = (AbstractPathTestSupport) testClass
220                     .getDeclaredConstructor().newInstance();
221             final String packageLocation = instance.getPackageLocation();
222 
223             scanExampleDirectory(examplesResources.resolve(packageLocation),
224                     testContent, testFile, failures);
225             scanExampleDirectory(examplesNonCompilable.resolve(packageLocation),
226                     testContent, testFile, failures);
227         }
228         catch (ReflectiveOperationException exception) {
229             throw new IllegalStateException("Failed to instantiate " + className, exception);
230         }
231     }
232 
233     private static void scanExampleDirectory(Path exampleDir, String testContent,
234             Path testFile, List<String> failures) throws IOException {
235         if (Files.exists(exampleDir) && Files.isDirectory(exampleDir)) {
236             try (Stream<Path> exampleFiles = Files.list(exampleDir)) {
237                 exampleFiles
238                     .filter(path -> {
239                         final String fileName = path.getFileName()
240                                 .toString();
241                         return fileName.matches("Example\\d+\\.java");
242                     })
243                     .forEach(exampleFile -> {
244                         final String fileName = exampleFile.getFileName()
245                                 .toString();
246                         if (!testContent.contains("\"" + fileName + "\"")) {
247                             failures.add("Missing test for " + fileName + " in "
248                                         + testFile.getFileName());
249                         }
250                     });
251             }
252         }
253     }
