22  import static com.google.common.truth.Truth.assertThat;
23  import static com.google.common.truth.Truth.assertWithMessage;
24  
25  import org.junit.jupiter.api.Test;
26  
27  import com.puppycrawl.tools.checkstyle.api.LineColumn;
28  import com.puppycrawl.tools.checkstyle.checks.javadoc.utils.TagInfo;
29  import com.puppycrawl.tools.checkstyle.utils.JavadocUtil;

31  public class JavadocTagTest {
32  
33      /* Additional test for jacoco, since valueOf()
34       * is generated by javac and jacoco reports that
35       * valueOf() is uncovered.
36       */
37      @Test
38      public void testJavadocTagTypeValueOf() {
39          final JavadocUtil.JavadocTagType enumConst =
40              JavadocUtil.JavadocTagType.valueOf("ALL");
41          assertWithMessage("Invalid enum valueOf result")
42              .that(enumConst)
43              .isEqualTo(JavadocUtil.JavadocTagType.ALL);
44      }
45  
46      /* Additional test for jacoco, since values()
47       * is generated by javac and jacoco reports that
48       * values() is uncovered.
49       */
50      @Test
51      public void testJavadocTagTypeValues() {
52          final JavadocUtil.JavadocTagType[] enumConstants =
53              JavadocUtil.JavadocTagType.values();
54          final JavadocUtil.JavadocTagType[] expected = {
55              JavadocUtil.JavadocTagType.BLOCK,
56              JavadocUtil.JavadocTagType.INLINE,
57              JavadocUtil.JavadocTagType.ALL,
58          };
59          assertWithMessage("Invalid enum constants")
60              .that(enumConstants)
61              .isEqualTo(expected);
62      }
63  
64      @Test
65      public void testToString() {
66          final JavadocTag javadocTag = new JavadocTag(0, 1, "author", "firstArg");
67  
68          final String result = javadocTag.toString();
69  
70          assertWithMessage("Invalid toString result")
71              .that(result)
72              .isEqualTo("JavadocTag[tag='author' lineNo=0, columnNo=1, firstArg='firstArg']");
73      }
74  
75      @Test
76      public void testJavadocTagReferenceImports() {
77          assertThat(new JavadocTag(0, 0, "see", null).canReferenceImports()).isTrue();
78          assertThat(new JavadocTag(0, 0, "link", null).canReferenceImports()).isTrue();
79          assertThat(new JavadocTag(0, 0, "value", null).canReferenceImports()).isTrue();
80          assertThat(new JavadocTag(0, 0, "linkplain", null).canReferenceImports()).isTrue();
81          assertThat(new JavadocTag(0, 0, "throws", null).canReferenceImports()).isTrue();
82          assertThat(new JavadocTag(0, 0, "exception", null).canReferenceImports()).isTrue();
83      }
84  
85      @Test
86      public void testJavadocTagReferenceImportsInvalid() {
87          assertThat(new JavadocTag(0, 0, "author", null).canReferenceImports())
88                  .isFalse();
89      }
90  
91      @Test
92      public void testJavadocTagIsInlineTag() {
93          assertThat(new JavadocTag(0, 0, "link", null).isInlineTag()).isTrue();
94          assertThat(new JavadocTag(0, 0, "value", null).isInlineTag()).isTrue();
95          assertThat(new JavadocTag(0, 0, "see", null).isInlineTag()).isFalse();
96  
97      }
98  
99      @Test
100     public void testInvalidJavadocTagLegacyGetters() {
101         final InvalidJavadocTag tag = new InvalidJavadocTag(1, 2, "test");
102         assertWithMessage("Invalid line number")
103             .that(tag.getLine())
104             .isEqualTo(1);
105         assertWithMessage("Invalid column number")
106             .that(tag.getCol())
107             .isEqualTo(2);
108         assertWithMessage("Invalid tag name")
109             .that(tag.getName())
110             .isEqualTo("test");
111         assertWithMessage("Invalid line number")
112             .that(tag.line())
113             .isEqualTo(1);
114         assertWithMessage("Invalid column number")
115             .that(tag.col())
116             .isEqualTo(2);
117         assertWithMessage("Invalid tag name")
118             .that(tag.name())
119             .isEqualTo("test");
120     }
121 
122     @Test
123     public void testTagInfoLegacyGetters() {
124         final LineColumn position = new LineColumn(1, 2);
125         final TagInfo tagInfo = new TagInfo("name", "value", position);
126         assertWithMessage("Invalid name")
127             .that(tagInfo.getName())
128             .isEqualTo("name");
129         assertWithMessage("Invalid value")
130             .that(tagInfo.getValue())
131             .isEqualTo("value");
132         assertWithMessage("Invalid position")
133             .that(tagInfo.getPosition())
134             .isEqualTo(position);
135         assertWithMessage("Invalid name")
136             .that(tagInfo.name())
137             .isEqualTo("name");
138         assertWithMessage("Invalid value")
139             .that(tagInfo.value())
140             .isEqualTo("value");
141         assertWithMessage("Invalid position")
142             .that(tagInfo.position())
143             .isEqualTo(position);
144     }
